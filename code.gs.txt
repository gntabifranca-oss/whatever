/**
 * Serves the form UI.
 * @return {HtmlOutput}
 */
function doGet() {
  return HtmlService.createHtmlOutputFromFile('index');
}

/**
 * Accepts form data and optional file attachment.
 *
 * @param {{name: string, email: string, message: string, attachment: {name: string, mimeType: string, data: string}|null}} payload
 * @return {{ok: boolean, message: string, attachmentId: string|null}}
 */
function submitForm(payload) {
  if (!payload || !payload.name || !payload.email) {
    throw new Error('Name and email are required.');
  }

  var attachmentId = null;

  if (payload.attachment && payload.attachment.data) {
    var bytes = Utilities.base64Decode(payload.attachment.data);
    var blob = Utilities.newBlob(
      bytes,
      payload.attachment.mimeType || 'application/octet-stream',
      payload.attachment.name || 'attachment'
    );
    var file = DriveApp.createFile(blob);
    attachmentId = file.getId();
  }

  Logger.log({
    name: payload.name,
    email: payload.email,
    message: payload.message || '',
    attachmentId: attachmentId,
  });

  return {
    ok: true,
    message: attachmentId
      ? 'Submission received and file uploaded.'
      : 'Submission received.',
    attachmentId: attachmentId,
  };
}
